const canvas = document.getElementById('game');
const ctx    = canvas.getContext('2d');

const scoreEl     = document.getElementById('score');
const bestEl      = document.getElementById('best');
const overlay     = document.getElementById('overlay');
const finalScoreEl = document.getElementById('final-score');
const restartBtn  = document.getElementById('restart');

const NEON_GREEN = '#39ff7e';
const NEON_HEAD  = '#a8ffcb';
const FOOD_COLOR = '#ff3f6c';
const GRID_COLOR = '#161616';
const GRID = 20;
const TICK = 110;

let COLS, ROWS;
let snake, dir, nextDir, food, score, best, running, loop;

// ── CANVAS RESPONSIVO ────────────────────────────────────────────────
function calcSize() {
  const maxH = window.innerHeight - 44 - 130;
  const maxW = window.innerWidth - 16;
  const size = Math.min(maxW, maxH, 420);
  return Math.floor(size / GRID) * GRID;
}

function resizeCanvas() {
  const size  = calcSize();
  canvas.width  = size;
  canvas.height = size;
  COLS = size / GRID;
  ROWS = size / GRID;
}

window.addEventListener('resize', () => { resizeCanvas(); draw(); });
resizeCanvas();

// ── INIT ─────────────────────────────────────────────────────────────
function init() {
  snake   = [{ x:10,y:10 },{ x:9,y:10 },{ x:8,y:10 }];
  dir     = { x:1, y:0 };
  nextDir = { x:1, y:0 };
  score   = 0;
  best    = parseInt(localStorage.getItem('snakeBest') || '0');
  scoreEl.textContent = 0;
  bestEl.textContent  = best;
  overlay.classList.add('hidden');
  placeFood();
  running = true;
  clearInterval(loop);
  loop = setInterval(tick, TICK);
}

function placeFood() {
  let pos;
  do {
    pos = { x: Math.floor(Math.random()*COLS), y: Math.floor(Math.random()*ROWS) };
  } while (snake.some(s => s.x===pos.x && s.y===pos.y));
  food = pos;
}

// ── GAME LOOP ─────────────────────────────────────────────────────────
function tick() {
  if (!running) return;
  dir = { ...nextDir };
  const head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

  if (head.x < 0 || head.x >= COLS || head.y < 0 || head.y >= ROWS) return gameOver();
  if (snake.some(s => s.x===head.x && s.y===head.y)) return gameOver();

  snake.unshift(head);

  if (head.x === food.x && head.y === food.y) {
    score++;
    scoreEl.textContent = score;
    if (score > best) {
      best = score;
      bestEl.textContent = best;
      localStorage.setItem('snakeBest', best);
    }
    placeFood();
  } else {
    snake.pop();
  }
  draw();
}

// ── DRAW ──────────────────────────────────────────────────────────────
function draw() {
  ctx.fillStyle = '#111';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.strokeStyle = GRID_COLOR;
  ctx.lineWidth = 0.5;
  for (let x = 0; x <= COLS; x++) {
    ctx.beginPath(); ctx.moveTo(x*GRID,0); ctx.lineTo(x*GRID,canvas.height); ctx.stroke();
  }
  for (let y = 0; y <= ROWS; y++) {
    ctx.beginPath(); ctx.moveTo(0,y*GRID); ctx.lineTo(canvas.width,y*GRID); ctx.stroke();
  }

  snake.forEach((seg, i) => {
    const isHead = i === 0;
    const m = isHead ? 1 : 2;
    ctx.fillStyle   = isHead ? NEON_HEAD : NEON_GREEN;
    ctx.shadowColor = isHead ? NEON_HEAD : NEON_GREEN;
    ctx.shadowBlur  = isHead ? 18 : 8;
    ctx.beginPath();
    ctx.roundRect(seg.x*GRID+m, seg.y*GRID+m, GRID-m*2, GRID-m*2, isHead ? 5 : 3);
    ctx.fill();
  });

  const fx = food.x*GRID + GRID/2;
  const fy = food.y*GRID + GRID/2;
  ctx.shadowColor = FOOD_COLOR;
  ctx.shadowBlur  = 20;
  ctx.fillStyle   = FOOD_COLOR;
  ctx.beginPath();
  ctx.arc(fx, fy, GRID/2 - 3, 0, Math.PI*2);
  ctx.fill();
  ctx.shadowBlur = 0;
}

function gameOver() {
  running = false;
  clearInterval(loop);
  finalScoreEl.textContent = `PONTUAÇÃO FINAL: ${score}`;
  overlay.classList.remove('hidden');
}

// ── TECLADO ───────────────────────────────────────────────────────────
const keyMap = {
  ArrowUp:    { x:0,  y:-1 },
  ArrowDown:  { x:0,  y: 1 },
  ArrowLeft:  { x:-1, y: 0 },
  ArrowRight: { x:1,  y: 0 },
};
document.addEventListener('keydown', e => {
  const d = keyMap[e.key];
  if (!d) return;
  if (d.x !== -dir.x || d.y !== -dir.y) nextDir = d;
  e.preventDefault();
});

// ── D-PAD ─────────────────────────────────────────────────────────────
function setDir(d) {
  if (!running) return;
  if (d.x !== -dir.x || d.y !== -dir.y) nextDir = d;
}

const dpadMap = {
  'btn-up':    { x:0,  y:-1 },
  'btn-down':  { x:0,  y: 1 },
  'btn-left':  { x:-1, y: 0 },
  'btn-right': { x:1,  y: 0 },
};
Object.entries(dpadMap).forEach(([id, d]) => {
  const btn = document.getElementById(id);
  btn.addEventListener('touchstart', e => { e.preventDefault(); setDir(d); }, { passive: false });
  btn.addEventListener('click', () => setDir(d));
});

// ── SWIPE NO CANVAS ───────────────────────────────────────────────────
let touchStart = null;
canvas.addEventListener('touchstart', e => {
  touchStart = { x: e.touches[0].clientX, y: e.touches[0].clientY };
}, { passive: true });
canvas.addEventListener('touchend', e => {
  if (!touchStart) return;
  const dx = e.changedTouches[0].clientX - touchStart.x;
  const dy = e.changedTouches[0].clientY - touchStart.y;
  if (Math.abs(dx) > Math.abs(dy)) {
    setDir(dx > 0 ? { x:1,y:0 } : { x:-1,y:0 });
  } else {
    setDir(dy > 0 ? { x:0,y:1 } : { x:0,y:-1 });
  }
  touchStart = null;
}, { passive: true });

// ── SERVICE WORKER ─────────────────────────────────────────────────────
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('./service-worker.js');
}

// ── START ──────────────────────────────────────────────────────────────
init();
draw();
